import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", }; Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } const respond = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); const url = Deno.env.get("SUPABASE_URL"); const key = Deno.env.get("MY_SECRET_KEY"); const adminPin = Deno.env.get("ADMIN_PIN"); if (!url || !key || !adminPin) { return respond({ error: "서버 설정이 필요합니다." }, 500); } const db = createClient(url, key); try { if (req.method === "GET") { const zone = new URL(req.url).searchParams.get("zone"); if (zone) { const { data, error } = await db .from("manual_sections") .select("id, content") .eq("id", zone) .maybeSingle(); if (error) throw error; return respond({ data }); } const { data, error } = await db .from("manual_sections") .select("id, content"); if (error) throw error; return respond({ data }); } if (req.method === "POST") { const body = await req.json(); if (typeof body.pin !== "string" || body.pin !== adminPin) { return respond({ error: "PIN이 올바르지 않습니다." }, 401); } // PIN verification request from the website. if (body.action === "verify") { return respond({ success: true }); } const allowedZones = [ "입구", "아쿠아&정글", "가재", "소라게", "게", "투구게", "새우", "출구", "후면 포스기", ]; if ( !allowedZones.includes(body.zone) || !body.content || typeof body.content !== "object" || Array.isArray(body.content) ) { return respond({ error: "잘못된 요청입니다." }, 400); } const { error } = await db .from("manual_sections") .upsert({ id: body.zone, content: body.content, updated_at: new Date().toISOString(), }); if (error) throw error; return respond({ success: true }); } return respond({ error: "지원하지 않는 요청입니다." }, 405); } catch (error) { console.error(error); return respond({ error: "데이터 처리에 실패했습니다." }, 500); } });
